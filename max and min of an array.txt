#include<stdio.h>
int main()
{
    int i,n,a[100],max,min;
    printf("enter no. of elements in array");
    scanf("%d",&n);
    for(i=0;i<n;i++)
        scanf("%d",&a[i]);
    max=a[0];
    for(i=0;i<n;i++)
        {
            if(max<a[i])
             max=a[i];   
        }
    printf("max=%d\n",max);
    min=a[0];
    for(i=0;i<n;i++)
        {
            if(min>a[i])
                min=a[i];
        }
    printf("min=%d",min);
    return 0;
}
